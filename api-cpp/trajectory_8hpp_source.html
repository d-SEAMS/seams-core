62[[nodiscard]] inline bool checkedMul(std::size_t a, std::size_t b,
63 std::size_t &out) noexcept {
64 if (a != 0 && b > std::numeric_limits<std::size_t>::max() / a) {
65 return false;
66 }
67 out = a * b;
68 return true;
