8// An exception that leaves an OpenMP region calls std::terminate, even when
9// the region runs on one thread or its if clause is false. Run each
10// iteration's body through run(): the first exception is kept, iterations
11// that start after it return at once, and rethrow() raises it once the
12// region has closed. The skipped iterations leave their outputs unwritten,
13// so a caller that catches the rethrow must not read them. The handler
14// itself cannot throw: a lock there would be one more std::terminate.
15class RegionErrors {
16public:
17 template <class F> void run(F &&body) noexcept {
18 if (failed_.load(std::memory_order_relaxed)) {
19 return;
20 }
21 try {
22 body();
23 } catch (...) {
24 if (!failed_.exchange(true, std::memory_order_acq_rel)) {
25 first_ = std::current_exception();
26 }
27 }
28 }
