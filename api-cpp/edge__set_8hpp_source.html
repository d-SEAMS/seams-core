17// Undirected edges, self entries dropped, each pair stored once as (lo, hi)

28void sortEdgePairs(std::vector<std::pair<int, int>> &edges) {
29#if !defined(SEAMS_HAS_OPENMP)
30 std::sort(edges.begin(), edges.end());
31#else
32 const std::size_t n = edges.size();
33 const int nT = omp_get_max_threads();
34 if (n < 8192 || nT < 2 || omp_in_parallel()) {
35 std::sort(edges.begin(), edges.end());
36 return;
37 }
38 const std::size_t chunk =
39 (n + static_cast<std::size_t>(nT) - 1) / static_cast<std::size_t>(nT);
40#pragma omp parallel for schedule(static)
41 for (int t = 0; t < nT; ++t) {
42 const std::size_t begin = static_cast<std::size_t>(t) * chunk;
43 if (begin >= n) {
44 continue;
45 }
46 const std::size_t end = std::min(n, begin + chunk);
47 std::sort(edges.begin() + static_cast<std::ptrdiff_t>(begin),
48 edges.begin() + static_cast<std::ptrdiff_t>(end));
49 }
50 for (std::size_t width = chunk; width < n; width *= 2) {
51 const std::size_t span = width * 2;
52 const int blocks =
53 static_cast<int>((n + span - 1) / span);
54#pragma omp parallel for schedule(static)
55 for (int b = 0; b < blocks; ++b) {
56 const std::size_t begin = static_cast<std::size_t>(b) * span;
57 const std::size_t mid = std::min(n, begin + width);
58 const std::size_t end = std::min(n, begin + span);
59 if (begin < mid && mid < end) {
60 std::inplace_merge(edges.begin() + static_cast<std::ptrdiff_t>(begin),
61 edges.begin() + static_cast<std::ptrdiff_t>(mid),
62 edges.begin() + static_cast<std::ptrdiff_t>(end));
63 }
64 }
65 }
66#endif
67}

69void hashEdges(EdgeSet &out) {
70 std::uint64_t h = 1469598103934665603ull;
71 for (const auto &e : out.edges) {
72 h ^= static_cast<std::uint64_t>(static_cast<std::uint32_t>(e.first));
73 h *= 1099511628211ull;
74 h ^= static_cast<std::uint64_t>(static_cast<std::uint32_t>(e.second));
75 h *= 1099511628211ull;
76 }
77 out.hash = h;
78}

98#pragma omp for schedule(static)

102 for (std::size_t k = 1; k < row.size(); ++k) {
103 const int j = row[k];
104 if (j < 0 || j == i) {
105 continue;

107 local.emplace_back(std::min(i, j), std::max(i, j));

116 total += part.size();

118 out.edges.reserve(total);
119 for (auto &part : parts) {
120 out.edges.insert(out.edges.end(), part.begin(), part.end());
121 }
122#endif
123 } else {
124 for (int i = 0; i < nRows; ++i) {
125 const auto &row = rows[static_cast<std::size_t>(i)];
126 for (std::size_t k = 1; k < row.size(); ++k) {
127 const int j = row[k];
128 if (j < 0 || j == i) {
