75inline void dumpBoundsToH(const std::vector<double> &box,
76 const std::vector<double> &boxLow, double H[3][3],
77 double origin[3]) {
78 const double xspan = box.size() > 0 ? box[0] : 0.0;
79 const double yspan = box.size() > 1 ? box[1] : 0.0;
80 const double zspan = box.size() > 2 ? box[2] : 0.0;
81 const double xlo_b = boxLow.size() > 0 ? boxLow[0] : 0.0;
82 const double ylo_b = boxLow.size() > 1 ? boxLow[1] : 0.0;
83 const double zlo_b = boxLow.size() > 2 ? boxLow[2] : 0.0;
84 const double xy = box.size() >= 6 ? box[3] : 0.0;
85 const double xz = box.size() >= 6 ? box[4] : 0.0;
86 const double yz = box.size() >= 6 ? box[5] : 0.0;
87 const double xmin = std::min(std::min(0.0, xy), std::min(xz, xy + xz));
88 const double xmax = std::max(std::max(0.0, xy), std::max(xz, xy + xz));
89 const double ymin = std::min(0.0, yz);
90 const double ymax = std::max(0.0, yz);
91 H[0][0] = xspan - xmax + xmin;
92 H[0][1] = 0.0;
93 H[0][2] = 0.0;
94 H[1][0] = xy;
95 H[1][1] = yspan - ymax + ymin;
96 H[1][2] = 0.0;
97 H[2][0] = xz;
98 H[2][1] = yz;
99 H[2][2] = zspan;
100 origin[0] = xlo_b - xmin;
101 origin[1] = ylo_b - ymin;
102 origin[2] = zlo_b;
103}

138inline void dumpToFrac(const double H[3][3], const double origin[3], double x,
139 double y, double z, double s[3]) {
140 const double lx = H[0][0];
141 const double ly = H[1][1];
142 const double lz = H[2][2];
143 const double xy = H[1][0];
144 const double xz = H[2][0];
145 const double yz = H[2][1];
146 s[2] = (z - origin[2]) / lz;
147 s[1] = (y - origin[1] - yz * s[2]) / ly;
148 s[0] = (x - origin[0] - xy * s[1] - xz * s[2]) / lx;
149}

152inline void dumpFromFrac(const double H[3][3], const double origin[3],
153 const double s[3], double r[3]) {
154 r[0] = origin[0] + H[0][0] * s[0] + H[1][0] * s[1] + H[2][0] * s[2];
155 r[1] = origin[1] + H[1][1] * s[1] + H[2][1] * s[2];
156 r[2] = origin[2] + H[2][2] * s[2];
157}

180inline bool setBoxFromLammpsEdges(std::vector<double> &box,
181 std::vector<double> &boxLow, double xlo,
182 double ylo, double zlo, double lx, double ly,
183 double lz, double xy, double xz, double yz) {
184 if (!(lx > 0.0 && ly > 0.0 && lz > 0.0)) {
185 box.clear();
186 boxLow.clear();
187 return false;
188 }
189 const bool tilt =
190 std::abs(xy) > 0.0 || std::abs(xz) > 0.0 || std::abs(yz) > 0.0;
191 if (!tilt) {
192 box = {lx, ly, lz};
193 boxLow = {xlo, ylo, zlo};
194 return true;
195 }
196 const double xmin = std::min(std::min(0.0, xy), std::min(xz, xy + xz));
197 const double xmax = std::max(std::max(0.0, xy), std::max(xz, xy + xz));
198 const double ymin = std::min(0.0, yz);
199 const double ymax = std::max(0.0, yz);
200 box = {lx + xmax - xmin, ly + ymax - ymin, lz, xy, xz, yz};
201 boxLow = {xlo + xmin, ylo + ymin, zlo};
202 return true;
203}

210inline bool setBoxFromLengthsAngles(std::vector<double> &box,
211 std::vector<double> &boxLow, double a,
212 double b, double c, double alphaDeg,
213 double betaDeg, double gammaDeg) {
214 constexpr double kDeg = 3.14159265358979323846 / 180.0;
215 const double alpha = alphaDeg * kDeg;
216 const double beta = betaDeg * kDeg;
217 const double gamma = gammaDeg * kDeg;
218 const double cosA = std::cos(alpha);
219 const double cosB = std::cos(beta);
220 const double cosG = std::cos(gamma);
221 const double sinG = std::sin(gamma);
222 if (!(a > 0.0 && b > 0.0 && c > 0.0) || std::abs(sinG) < 1e-18) {
223 box.clear();
224 boxLow.clear();
225 return false;
226 }
227 double xy = b * cosG;
228 const double ly = b * sinG;
229 double xz = c * cosB;
230 double yz = c * (cosA - cosB * cosG) / sinG;
231 // cos(90 deg) is a few ulps, not zero. A right angle must stay
232 // orthorhombic, or the pair loop takes the triclinic path.
233 const double tol = 1e-10 * std::max(a, std::max(b, c));
234 if (std::abs(xy) < tol) {
235 xy = 0.0;
236 }
237 if (std::abs(xz) < tol) {
238 xz = 0.0;
239 }
240 if (std::abs(yz) < tol) {
241 yz = 0.0;
242 }
243 const double lz2 = c * c - xz * xz - yz * yz;
244 if (!(ly > 0.0) || !(lz2 > 0.0)) {
245 box.clear();
246 boxLow.clear();
247 return false;
248 }
249 return setBoxFromLammpsEdges(box, boxLow, 0.0, 0.0, 0.0, a, ly, std::sqrt(lz2),
250 xy, xz, yz);
251}

255inline lc_cell lammpsBoxToLcCell(const std::vector<double> &box,
256 const std::vector<double> &boxLow) {
257#ifdef SEAMS_HAS_MINIMAGE
258 const double xspan = box.size() > 0 ? box[0] : 0.0;
259 const double yspan = box.size() > 1 ? box[1] : 0.0;
260 const double zspan = box.size() > 2 ? box[2] : 0.0;
261 const double xy = box.size() >= 6 ? box[3] : 0.0;
262 const double xz = box.size() >= 6 ? box[4] : 0.0;
263 const double yz = box.size() >= 6 ? box[5] : 0.0;
264 const double xlo_b = boxLow.size() > 0 ? boxLow[0] : 0.0;
265 const double ylo_b = boxLow.size() > 1 ? boxLow[1] : 0.0;
266 const double zlo_b = boxLow.size() > 2 ? boxLow[2] : 0.0;
267 mi_cell raw{};
268 if (mi_cell_from_lammps_bounds(xspan, yspan, zspan, xy, xz, yz, xlo_b, ylo_b,
269 zlo_b, &raw) == 0) {
270 lc_cell c = lc_cell_ortho(raw.ax, raw.by, raw.cz);
271 c.ax = raw.ax;
272 c.ay = raw.ay;
273 c.az = raw.az;
274 c.bx = raw.bx;
275 c.by = raw.by;
276 c.bz = raw.bz;
277 c.cx = raw.cx;
278 c.cy = raw.cy;
279 c.cz = raw.cz;
280 c.ox = raw.ox;
281 c.oy = raw.oy;
282 c.oz = raw.oz;
283 return c;
284 }
285#endif
286 double H[3][3], o[3];
287 dumpBoundsToH(box, boxLow, H, o);
288 lc_cell c = lc_cell_ortho(H[0][0], H[1][1], H[2][2]);
289 c.ax = H[0][0];
290 c.ay = H[0][1];
291 c.az = H[0][2];
292 c.bx = H[1][0];
293 c.by = H[1][1];
294 c.bz = H[1][2];
295 c.cx = H[2][0];
296 c.cy = H[2][1];
297 c.cz = H[2][2];
298 c.ox = o[0];
299 c.oy = o[1];
300 c.oz = o[2];
301 return c;
302}

313inline void residentFrameCell(const double *box, const double *boxLow,
314 int nBox, linkcell::Cell &cell,
315 const double *&frameLens) {
316 cell = linkcell::Cell::ortho(box[0], box[1], box[2]);
317 frameLens = box;
318 if (boxLow != nullptr || nBox >= 6) {
319 std::vector<double> dump(static_cast<std::size_t>(std::max(nBox, 3)));
320 for (int i = 0; i < nBox && i < static_cast<int>(dump.size()); ++i) {
321 dump[static_cast<std::size_t>(i)] = box[i];
322 }
323 std::vector<double> lo;
324 if (boxLow != nullptr) {
325 lo = {boxLow[0], boxLow[1], boxLow[2]};
326 }
327 cell = lammpsBoxToLinkcell(dump, lo);
328 frameLens = nullptr;
329 }
