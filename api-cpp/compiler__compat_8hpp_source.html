4// GNU target attributes and the inline xgetbv stay on GCC and Clang,
5// including clang-cl. MSVC has no target attribute; those kernels take
6// the scalar twin, which uses the same operation order.
7
8#if defined(__GNUC__) || defined(__clang__)
9#define SEAMS_TARGET(spec) __attribute__((target(spec)))
10// Clang rejects the GCC optimize attribute. The kernels keep their
11// default optimisation there; GCC still requests the named pass.
12#if defined(__clang__)
13#define SEAMS_OPTIMIZE(spec)
14#else
15#define SEAMS_OPTIMIZE(spec) __attribute__((optimize(spec)))
16#endif
17#define SEAMS_WEAK __attribute__((weak))
18#define SEAMS_NOINLINE __attribute__((noinline))
19#define SEAMS_ALWAYS_INLINE __attribute__((always_inline)) inline
20#define SEAMS_COLD __attribute__((cold))
21#define SEAMS_EXPORT_USED __attribute__((visibility("default"), used))
22#define SEAMS_GNU_TARGET 1
23#else
24#define SEAMS_TARGET(spec)
25#define SEAMS_OPTIMIZE(spec)
26#define SEAMS_WEAK
27#define SEAMS_NOINLINE
28#define SEAMS_ALWAYS_INLINE __forceinline
29#define SEAMS_COLD
30#define SEAMS_EXPORT_USED

33// Exception types that cross from libyodaLib into libseams. libc++ matches
34// a handler by the address of typeinfo, and libseams is built with hidden
35// visibility, so the key function is one default-visibility symbol in
36// libyodaLib. On Windows, SEAMS_EH_EXPORT dllexports the class from
37// libyodaLib and every other translation unit dllimports it. MSVC matches
38// the decorated type name, so those addresses are not one value.
39#if defined(_WIN32)
40#if defined(SEAMS_EH_EXPORT)
41#define SEAMS_EH_API __declspec(dllexport)
42#else
43#define SEAMS_EH_API __declspec(dllimport)
44#endif
45#elif defined(__GNUC__) || defined(__clang__)
46#define SEAMS_EH_API __attribute__((visibility("default")))
47#else
48#define SEAMS_EH_API
49#endif
