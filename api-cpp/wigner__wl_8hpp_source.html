226template <int L, std::size_t N>
227double contract(const double *qlm, const Term (&terms)[N]) {
228 const int nComp = 2 * L + 1;
229 double norm2 = 0.0;
230 for (int m = 0; m < nComp; ++m) {
231 const double re = qlm[2 * m];
232 const double im = qlm[2 * m + 1];
233 norm2 += re * re + im * im;
234 }
235 if (!(norm2 > 0.0)) {
236 return 0.0;
237 }
238 double acc = 0.0;
239 for (const Term &t : terms) {
240 const double a = qlm[2 * t.i1];
241 const double b = qlm[2 * t.i1 + 1];
242 const double c = qlm[2 * t.i2];
243 const double d = qlm[2 * t.i2 + 1];
244 const double e = qlm[2 * t.i3];
245 const double f = qlm[2 * t.i3 + 1];
246 const double re = a * (c * e - d * f) - b * (c * f + d * e);
247 acc += t.w * re;
248 }
249 return acc / (norm2 * std::sqrt(norm2));
250}
