4// libgomp's barrier does not publish a happens-before that ThreadSanitizer
5// can see, so a write inside a parallel region and a read after the join

7// the sanitizer instruments. It is empty on every other build.

23inline void arrive_and_wait(Slot &slot, int threads) {
24#ifdef SEAMS_TSAN_JOIN

33 return;
34 }

36 }
