193inline molSys::bond_type classifyBond(double cij, const BondClassifier &rule,
194 bool closed = true) {
195 const bool staggered = closed ? cij <= rule.staggeredMax : cij < rule.staggeredMax;
196 if (staggered) {
197 return molSys::bond_type::staggered;
198 }
199 const bool eclipsed =
200 closed ? (cij >= rule.eclipsedMin && cij <= rule.eclipsedMax)
201 : (cij > rule.eclipsedMin && cij < rule.eclipsedMax);
202 if (eclipsed) {
203 return molSys::bond_type::eclipsed;
204 }
205 return molSys::bond_type::out_of_range;
206}
