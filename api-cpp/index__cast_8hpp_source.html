15// or count that cannot be negative, or a size that must fit an int. A

17// negative count. Not for hot loops; those keep their own index type.
18template <std::integral To, std::integral From> constexpr To checkedCast(From value) {
19 if (!std::in_range<To>(value)) {
20 throw std::out_of_range("integer value out of range for its target type");
21 }
22 return static_cast<To>(value);
23}
