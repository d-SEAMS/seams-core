19// Stages of chunks in one parallel region. A run claims each chunk once
20// and every chunk of a stage returns before the next stage starts.
21//
22// While other processes keep half of the caller's cpus busy or more, the
23// region runs on linkcell's pool, the caller one of the workers: a chunk
24// is claimed from a counter and a stage closes when its last claimed
25// chunk returns, so neither a stage boundary nor the return waits for a
26// worker the scheduler has not run. An OpenMP join or barrier waits for
27// every thread of the team: beside an eight-thread spinner on eight cpus
28// one join of an empty loop took 3 to 11 ms, and a q6 call of 4 ms took
29// 8 to 19 ms in the processes whose team met one. On a machine the caller
30// has to itself, the region is one OpenMP team with a loop per stage,
31// whose spinning workers start each stage at once; Rayon workers that
32// have gone to sleep do not, and the q6 call took 3.1 ms on them instead
33// of 2.3.

36// Workers a run started now may use. The `worker` index a run passes is
37// below it, so it sizes per-worker scratch. Linkcell's team is the pool
38// run() selects with lc_set_threads(poolFor(0)), so it is never wider
39// than poolFor(0); asking lc_team_width would start that pool on a
40// machine where the run stays on OpenMP.
41inline int width() {
42#ifdef SEAMS_HAS_OPENMP
43 if (omp_in_parallel()) {
44 return 1;
45 }
46#endif
47 return std::max(teamWidth(), poolFor(0));
48}

50// Chunks per stage for `width` workers: one per OpenMP thread on a
51// machine the caller has to itself, so each thread keeps one share as the
52// original loops did, and two per worker while other processes crowd the
53// cpus, so a worker the scheduler parks holds up a smaller share. The
54// OpenMP team can be narrower than `width`, which also covers linkcell's
55// pool: eight chunks on a team narrowed to seven gave one thread two, and
56// the 110592-molecule cutoff list took 2.8 ms instead of 1.5.
57inline int parts(int width) {
58 if (width <= 1) {
59 return 1;
60 }
61 return othersHoldHalf() ? 2 * width : std::max(1, std::min(width, teamWidth()));
62}

66// The handler runs inside the region or a pool worker and cannot throw:
67// the first failure claims the slot with an atomic exchange, not a lock.
68template <class F> struct Call {
69 F *fn;
70 std::atomic<bool> failed{false};
71 std::exception_ptr failure;

73 void one(std::size_t stage, std::size_t chunk, std::size_t worker) noexcept {
74 if (failed.load(std::memory_order_relaxed)) {
75 return;
76 }
77 try {
78 (*fn)(stage, chunk, static_cast<int>(worker));
79 } catch (...) {
80 if (!failed.exchange(true, std::memory_order_acq_rel)) {
81 failure = std::current_exception();
82 }
83 }
84 }

92template <class C> void ompRun(const std::vector<std::size_t> &sizes, C &call) {
93 std::size_t most = 0;
94 for (const std::size_t n : sizes) {
95 most = std::max(most, n);
96 }
97#ifdef SEAMS_HAS_OPENMP
98 const int threads =
99 omp_in_parallel() ? 1 : static_cast<int>(std::min<std::size_t>(
100 static_cast<std::size_t>(omp_get_max_threads()), most));
101 if (threads > 1) {
102#pragma omp parallel num_threads(threads)
103 {
104 const std::size_t worker = static_cast<std::size_t>(omp_get_thread_num());
105 for (std::size_t s = 0; s < sizes.size(); ++s) {
106 const long long count = static_cast<long long>(sizes[s]);
107 // Each thread takes one contiguous run of chunks, as the loops
108 // this replaced did.
109#pragma omp for schedule(static)
110 for (long long c = 0; c < count; ++c) {
111 call.one(s, static_cast<std::size_t>(c), worker);
112 }
113 }
114 }
115 return;
116 }
117#endif
118 for (std::size_t s = 0; s < sizes.size(); ++s) {
119 for (std::size_t c = 0; c < sizes[s]; ++c) {
120 call.one(s, c, 0);
121 }
122 }
123}

127// Calls fn(stage, chunk, worker) once for every chunk of every stage.
128// Every chunk of a stage returns before any chunk of the next starts.
129// The first exception is rethrown once the region ends; chunks not yet
130// started when it was thrown are skipped.
131template <class F> void run(const std::vector<std::size_t> &sizes, F &&fn) {
132 if (sizes.empty()) {
133 return;
134 }
135 detail::Call<std::remove_reference_t<F>> call;
136 call.fn = &fn;
137 bool nested = false;
138#ifdef SEAMS_HAS_OPENMP
139 nested = omp_in_parallel() != 0;
140#endif
141 if (!nested && othersHoldHalf()) {
142 lc_set_threads(poolFor(0));
143 if (lc_team_run(sizes.size(), sizes.data(), &decltype(call)::trampoline, &call) != 0) {
144 throw std::runtime_error("linkcell team run failed");
145 }
146 } else {
147 detail::ompRun(sizes, call);
148 }
149 if (call.failure) {
150 std::rethrow_exception(call.failure);
151 }
152}

158// Chunks of `grain` items covering 0..n.
159inline std::size_t chunksOf(std::size_t n, std::size_t grain) {
160 return grain == 0 ? 0 : (n + grain - 1) / grain;
161}

163// One stage over 0..n in chunks of about `grain`: fn(begin, end,
164// worker). The chunk count is rounded up to a multiple of the workers
165// the run starts, so each takes the same number of equal chunks.
166template <class F> void forRange(std::size_t n, std::size_t grain, F &&fn) {
167 if (n == 0) {
168 return;
169 }
170 std::size_t count = chunksOf(n, std::max<std::size_t>(1, grain));
171 const int most = width();
172 const auto workers = static_cast<std::size_t>(
173 othersHoldHalf() ? most : std::max(1, std::min(most, teamWidth())));
174 if (count > 1 && workers > 1) {
175 count = std::min(n, (count + workers - 1) / workers * workers);
176 }
177 const std::size_t step = chunksOf(n, count);
178 run({chunksOf(n, step)}, [&](std::size_t, std::size_t chunk, int worker) {
179 const std::size_t begin = chunk * step;
180 fn(begin, std::min(n, begin + step), worker);
181 });
182}
