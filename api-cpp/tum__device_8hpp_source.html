148 int kMax, const int *b1, const int *b2) {
149 const int perm[6][3] = {{0, 1, 2}, {0, 2, 1}, {1, 0, 2},
150 {1, 2, 0}, {2, 0, 1}, {2, 1, 0}};
151 for (int q = 0; q < 2; ++q) {

172inline int ringsThrough(const int *A, int nA, const int *B, int nB,
173 const int *C, int nC, int skipA, int skipB, int *out,
174 int cap) {
175 int i = 0;
176 int j = 0;
177 int k = 0;
178 int n = 0;
179 while (i < nA && j < nB && k < nC) {
180 const int x = A[i];
181 const int y = B[j];
182 const int z = C[k];
183 if (x == y && y == z) {
184 if (x != skipA && x != skipB && n < cap) {
185 out[n++] = x;
186 }
187 ++i;
188 ++j;
189 ++k;
190 continue;
191 }
192 int lo = x;
193 if (y < lo) {
194 lo = y;
195 }
196 if (z < lo) {
197 lo = z;
198 }
199 if (x == lo) {
200 ++i;
201 }
202 if (y == lo) {
203 ++j;
204 }
205 if (z == lo) {
206 ++k;
207 }
208 }
209 return n;
210}

225inline void enumSixFrom(int i, const int *deg, const int *cols, int nAtoms,
226 int kMax, int maxRings, int *nRings, int *ringAtoms) {
227 const int di = deg[i];
228 if (di < 2) {
229 return;
230 }
231 const int irow = i * kMax;
232 for (int ia = 0; ia < di; ++ia) {
233 const int a = cols[irow + ia];
234 for (int ib = ia + 1; ib < di; ++ib) {
235 const int b = cols[irow + ib];
236 const int da = deg[a];
237 const int db = deg[b];
238 const int arow = a * kMax;
239 const int brow = b * kMax;
240 for (int ix = 0; ix < da; ++ix) {
241 const int x = cols[arow + ix];
242 if (x == i || x == b) {
243 continue;
244 }
245 for (int iy = 0; iy < db; ++iy) {
246 const int y = cols[brow + iy];
247 if (y == i || y == a || y == x) {
248 continue;
249 }
250 const int dx = deg[x];
251 const int xrow = x * kMax;
252 for (int iz = 0; iz < dx; ++iz) {
253 const int z = cols[xrow + iz];
254 if (z == i || z == a || z == b || z == y) {
255 continue;
256 }
257 if (!bonded(deg, cols, nAtoms, kMax, z, y)) {
258 continue;
259 }
260 const int cyc[6] = {i, a, x, z, y, b};
261 int mn = i;
262 for (int t = 1; t < 6; ++t) {
263 if (cyc[t] < mn) {
264 mn = cyc[t];
265 }
266 }
267 if (mn != i) {
268 continue;
269 }
270 if (!hopBoundPrimitiveSix(cyc, deg, cols, nAtoms, kMax)) {
271 continue;
272 }
273 const int slot = fetchAdd(nRings);
274 if (slot >= maxRings) {
275 continue;
276 }
277 const int dest = slot * 6;
278 for (int t = 0; t < 6; ++t) {
279 ringAtoms[dest + t] = cyc[t];
280 }
281 }
282 }
283 }
284 }

404inline void applyHcPair(int p, const int *nPairs, const int *pairs,
405 const int *ringAtoms, const int *throughCount,
406 const int *through, int nAtoms, int maxPer, int *hc) {
407 if (p >= *nPairs) {
408 return;
409 }
410 const int i = pairs[p * 2];
411 const int j = pairs[p * 2 + 1];
412 hc[i] = 1;
413 hc[j] = 1;
414 const int *bi = ringAtoms + i * 6;
415 const int *bj = ringAtoms + j * 6;
416 for (int q = 0; q < 6; ++q) {
417 int trip[3];
418 for (int m = 0; m < 3; ++m) {
419 trip[m] = bi[(q + m) % 6];
420 }
421 const int nA = throughCount[trip[0]];
422 const int nB = throughCount[trip[1]];
423 const int nC = throughCount[trip[2]];

426 const int *C = through + trip[2] * maxPer;
427 int cand[kRowCap];
428 const int nc = ringsThrough(A, nA, B, nB, C, nC, i, j, cand, kRowCap);
429 for (int c = 0; c < nc; ++c) {
430 const int kr = cand[c];
431 const int *bk = ringAtoms + kr * 6;
432 int rest = 0;
433 int restInJ = 0;
434 for (int u = 0; u < 6; ++u) {
435 const int a = bk[u];
436 if (a != trip[0] && a != trip[1] && a != trip[2]) {
437 ++rest;
438 if (inSix(bj, a)) {

441 }

455inline void ddcFrom(int i, const int *nRings, const int *ringAtoms,
456 const int *throughCount, const int *through, const int *hc,
457 int nAtoms, int maxPer, int *ddc) {
458 if (i >= *nRings) {
459 return;
460 }
461 if (hc[i]) {
462 return;
463 }
464 const int *bi = ringAtoms + i * 6;
465 for (int m = 0; m < 6; ++m) {
466 const int atom = bi[m];
467 if (atom < 0 || atom >= nAtoms) {
468 return;
469 }
470 const int nr = throughCount[atom];
471 const int *row = through + atom * maxPer;
472 int common = 0;
473 for (int t = 0; t < nr; ++t) {
474 if (row[t] != i) {
475 ++common;
476 }
477 }
478 if (common < 3) {
479 return;
480 }
481 }
482 // T[k]: every ring but i through bi[k], bi[k+1], bi[k+2]
483 int T[6][kRowCap];
484 int nT[6];
485 for (int k = 0; k < 6; ++k) {
486 int trip[3];
487 for (int t = 0; t < 3; ++t) {
488 trip[t] = bi[(k + t) % 6];
489 }
490 const int nA = throughCount[trip[0]];
491 const int nB = throughCount[trip[1]];
492 const int nC = throughCount[trip[2]];
493 const int *A = through + trip[0] * maxPer;
494 const int *B = through + trip[1] * maxPer;
495 const int *C = through + trip[2] * maxPer;
496 nT[k] = ringsThrough(A, nA, B, nB, C, nC, i, -1, T[k], kRowCap);
497 if (nT[k] == 0) {
498 return;
499 }
500 }
501 // Bit t of used[k] marks T[k][t] as P_k of some valid assignment
502 unsigned used[6] = {0U, 0U, 0U, 0U, 0U, 0U};
503 bool any = false;
504 for (int a = 0; a < nT[0]; ++a) {
505 const int r0 = T[0][a];
506 const int *p0 = ringAtoms + r0 * 6;
507 for (int c = 0; c < nT[2]; ++c) {
508 const int r2 = T[2][c];
509 const int *p2 = ringAtoms + r2 * 6;
510 if (r2 == r0) {
511 continue;
512 }
513 const bool e02 = commonCount(p0, p2) >= 3;
514 for (int e = 0; e < nT[4]; ++e) {
515 const int r4 = T[4][e];
516 const int *p4 = ringAtoms + r4 * 6;
517 if (r4 == r0 || r4 == r2 || !commonInThree(p0, p2, p4)) {
518 continue;
519 }
520 const bool e24 = commonCount(p2, p4) >= 3;
521 const bool e40 = commonCount(p4, p0) >= 3;
522 // Bit t: the ring on triplet 2t shares 3 atoms with both others
523 const unsigned mE = (e40 && e02 ? 1U : 0U) |
524 (e02 && e24 ? 2U : 0U) |
525 (e24 && e40 ? 4U : 0U);
526 if (mE == 0U) {
527 continue;
528 }
529 // Some start and direction of i puts an even middle 2t next to an
530 // odd middle 2t+1 or 2t-1
531 unsigned partners = 0U;
532 for (int t = 0; t < 3; ++t) {
533 if ((mE >> t) & 1U) {
534 partners |= (1U << t) | (1U << ((t + 2) % 3));
535 }
536 }
537 for (int b = 0; b < nT[1]; ++b) {
538 const int r1 = T[1][b];
539 const int *p1 = ringAtoms + r1 * 6;
540 if (r1 == r0 || r1 == r2 || r1 == r4) {
541 continue;
542 }
543 for (int d = 0; d < nT[3]; ++d) {
544 const int r3 = T[3][d];
545 const int *p3 = ringAtoms + r3 * 6;
546 if (r3 == r0 || r3 == r2 || r3 == r4 || r3 == r1) {
547 continue;
548 }
549 const bool o13 = commonCount(p1, p3) >= 3;
550 for (int g = 0; g < nT[5]; ++g) {
551 const int r5 = T[5][g];
552 const int *p5 = ringAtoms + r5 * 6;
553 if (r5 == r0 || r5 == r2 || r5 == r4 || r5 == r1 || r5 == r3 ||
554 !commonInThree(p1, p3, p5)) {
555 continue;
556 }
557 const bool o35 = commonCount(p3, p5) >= 3;
558 const bool o51 = commonCount(p5, p1) >= 3;
559 const unsigned mO = (o51 && o13 ? 1U : 0U) |
560 (o13 && o35 ? 2U : 0U) |
561 (o35 && o51 ? 4U : 0U);
562 if ((partners & mO) == 0U) {
563 continue;
564 }
565 any = true;
566 used[0] |= 1U << a;
567 used[1] |= 1U << b;
568 used[2] |= 1U << c;
569 used[3] |= 1U << d;
570 used[4] |= 1U << e;
571 used[5] |= 1U << g;
572 }
573 }
574 }
575 }
576 }
577 }
578 if (!any) {
579 return;
580 }
581 ddc[i] = 1;
582 for (int k = 0; k < 6; ++k) {
583 for (int t = 0; t < nT[k]; ++t) {
584 if ((used[k] >> t) & 1U) {
585 ddc[T[k][t]] = 1;
586 }
587 }
588 }
