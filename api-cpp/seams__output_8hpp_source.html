36[[nodiscard]] inline int makePath(const std::string &path) {
37 std::error_code ec;
38 if (fs::is_directory(path)) {
39 return 0;
40 }
41 if (fs::create_directories(path, ec)) {
42 return 0;
43 }
44 return ec ? 1 : 0;
45}
