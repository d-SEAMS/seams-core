32// Associated-Legendre amplitude without the azimuthal factor, in the
33// hard-coded table convention (the sin^|m| factor is carried by w^m).
34template <class T>
35inline T legendrePoly(int orderL, int absM, const T *zpow) {
36 if (orderL == 3) {
37 switch (absM) {
38 case 0:
39 return T(5) * zpow[3] - T(3) * zpow[1];
40 case 1:
41 return T(5) * zpow[2] - T(1);
42 case 2:
43 return zpow[1];
44 default:
45 return T(1);
46 }
47 }
48 switch (absM) {
49 case 0:
50 return T(231) * zpow[6] - T(315) * zpow[4] + T(105) * zpow[2] - T(5);
51 case 1:
52 return T(33) * zpow[5] - T(30) * zpow[3] + T(5) * zpow[1];
53 case 2:
54 return T(33) * zpow[4] - T(18) * zpow[2] + T(1);
55 case 3:
56 return T(11) * zpow[3] - T(3) * zpow[1];
57 case 4:
58 return T(11) * zpow[2] - T(1);
59 case 5:
60 return zpow[1];
61 default:
62 return T(1);
63 }
64}

66// x, y, z are components of a unit vector. re/im hold 2*l+1 values,
67// m = -l .. +l. l is 3 or 6.
68template <class T>
69inline void evalUnit(int orderL, T x, T y, T z, T *re, T *im) {
70 T wre[7];
71 T wim[7];
72 T zpow[7];
73 wre[0] = T(1);
74 wim[0] = T(0);
75 zpow[0] = T(1);
76 for (int m = 1; m <= orderL; ++m) {
77 const T prevRe = wre[m - 1];
78 const T prevIm = wim[m - 1];
79 wre[m] = prevRe * y - prevIm * x;
80 wim[m] = prevRe * x + prevIm * y;
81 zpow[m] = zpow[m - 1] * z;
82 }
83 const double *norm = (orderL == 3) ? N3 : N6;
84 for (int m = 0; m <= orderL; ++m) {
85 const T amp = T(norm[m]) * legendrePoly(orderL, m, zpow);
86 const int ineg = orderL - m;
87 const int ipos = orderL + m;
88 re[ineg] = amp * wre[m];
89 im[ineg] = -amp * wim[m];
90 const T signedAmp = (m & 1) ? -amp : amp;
91 re[ipos] = signedAmp * wre[m];
92 im[ipos] = signedAmp * wim[m];
93 }
94}

96// Displacement, not necessarily a unit vector. A zero vector writes
97// zeros and returns false so the caller can drop the bond.
98inline bool ylmReal(int orderL, double dx, double dy, double dz, double *re,
99 double *im) {
100 const int nComp = 2 * orderL + 1;
101 const double r2 = dx * dx + dy * dy + dz * dz;
102 if (!(r2 > 0.0) || (orderL != 3 && orderL != 6)) {
103 for (int m = 0; m < nComp; ++m) {
104 re[m] = 0.0;
105 im[m] = 0.0;
106 }
107 return false;
108 }
109 const double inv = 1.0 / std::sqrt(r2);
110 evalUnit(orderL, dx * inv, dy * inv, dz * inv, re, im);
111 return true;
112}
