108void parallelStableSort(std::vector<T> &values, Less less) {
109 const int n = static_cast<int>(values.size());
110 const int threads = teamThreads();
111 if (threads <= 1 || n < threads * 2) {
112 std::stable_sort(values.begin(), values.end(), less);
113 return;
114 }
115 const int chunk = (n + threads - 1) / threads;
116#ifdef SEAMS_HAS_OPENMP
117#pragma omp parallel for schedule(static)
118#endif
119 for (int tid = 0; tid < threads; tid++) {
120 const int lo = tid * chunk;
121 const int hi = std::min(n, lo + chunk);
122 if (hi - lo > 1) {
123 std::stable_sort(values.begin() + lo, values.begin() + hi, less);
124 }
125 }

128 T *dst = buffer.data();
129 for (int width = chunk; width < n; width *= 2) {
130#ifdef SEAMS_HAS_OPENMP

132#endif

134 const int mid = std::min(n, start + width);

136 int i = start;
137 int j = mid;
138 int k = start;
139 while (i < mid && j < stop) {
140 if (less(src[j], src[i])) {
