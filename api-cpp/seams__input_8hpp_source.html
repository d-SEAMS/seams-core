246inline bool atomInSlice(double x, double y, double z,
247 std::array<double, 3> coordLow,
248 std::array<double, 3> coordHigh) {
249 int flag = 0;
250
251 if (((x >= coordLow[0]) && (x <= coordHigh[0])) ||
252 coordLow[0] == coordHigh[0]) {
253 flag++;
254 }
255 if (((y >= coordLow[1]) && (y <= coordHigh[1])) ||
256 coordLow[1] == coordHigh[1]) {
257 flag++;
258 }
259 if (((z >= coordLow[2]) && (z <= coordHigh[2])) ||
260 coordLow[2] == coordHigh[2]) {
261 flag++;
262 }
263
264 if (flag == 3) {
265 return true;
266 } else {
267 return false;
268 }
269}
