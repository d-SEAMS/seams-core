224inline bool atomInSlice(double x, double y, double z,
225 std::array<double, 3> coordLow,
226 std::array<double, 3> coordHigh) {
227 int flag = 0;
228
229 if (((x >= coordLow[0]) && (x <= coordHigh[0])) ||
230 coordLow[0] == coordHigh[0]) {
231 flag++;
232 }
233 if (((y >= coordLow[1]) && (y <= coordHigh[1])) ||
234 coordLow[1] == coordHigh[1]) {
235 flag++;
236 }
237 if (((z >= coordLow[2]) && (z <= coordHigh[2])) ||
238 coordLow[2] == coordHigh[2]) {
239 flag++;
240 }
241
242 if (flag == 3) {
243 return true;
244 } else {
245 return false;
246 }
247}
