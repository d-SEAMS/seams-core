13// One width for OpenMP and Rayon. OpenMP already follows
14// OMP_NUM_THREADS. Linkcell's pool does not, so a process whose
15// hardware reports one cpu keeps the neighbour walk on one thread
16// even when OpenMP was given several. lc_set_threads is read by the
17// next linkcell search.
18//
19// OMP_PROC_BIND makes the OpenMP runtime pin the calling thread to its
20// first place when the library loads, and threads created afterwards, a
21// Rayon pool among them, inherit that one cpu. widenAffinity gives the
22// caller back the cpus of every place, which the runtime took from the
23// launch mask (taskset, numactl, a cgroup cpuset) and OMP_PLACES. It adds
24// no cpu outside those places. A caller the runtime did not bind, and a
25// one-thread width, keep their mask. The pool is built once and its
26// workers sleep between searches. The OpenMP team is not paused: a pause
27// makes the next region pay a wake-up that moves by tens of milliseconds.

47inline void widenAffinity() {
48#if defined(__linux__) && defined(SEAMS_HAS_OPENMP)
49 static std::mutex lock;
50 static bool done = false;
51 if (omp_in_parallel() || omp_get_max_threads() < 2 || omp_get_place_num() < 0) {
52 return;
53 }
54 const std::lock_guard<std::mutex> hold(lock);
55 if (done) {
56 return;
57 }
58 done = true;
59 cpu_set_t mask;
60 CPU_ZERO(&mask);
61 if (sched_getaffinity(0, sizeof(mask), &mask) != 0) {
62 return;
63 }
64 const int places = omp_get_num_places();
65 for (int place = 0; place < places; ++place) {
66 const int count = omp_get_place_num_procs(place);
67 if (count <= 0) {
68 continue;
69 }
70 std::vector<int> ids(static_cast<std::size_t>(count));
71 omp_get_place_proc_ids(place, ids.data());
72 for (const int cpu : ids) {
73 if (cpu >= 0 && cpu < CPU_SETSIZE) {
74 CPU_SET(static_cast<std::size_t>(cpu), &mask);
75 }
76 }
77 }
78 sched_setaffinity(0, sizeof(mask), &mask);

82// glibc hands the top of the heap back to the kernel past M_TRIM_THRESHOLD
83// and serves blocks above M_MMAP_THRESHOLD with fresh mappings. Both start
84// low and move with the allocation pattern, so the buffers one call freed
85// were faulted in again by the next call, by an amount that changed from
86// run to run. glibc has no per-library heap and these settings hold for
87// the whole process. SEAMS_MALLOC_TUNE=0 leaves them alone, and so does a
88// malloc setting the caller already made through MALLOC_*_ or
89// GLIBC_TUNABLES.
90inline void tuneHeap() {
91#ifdef __GLIBC__
92 static const bool tuned = [] {
93 const char *opt = std::getenv("SEAMS_MALLOC_TUNE");
94 if (opt != nullptr && (std::strcmp(opt, "0") == 0 || std::strcmp(opt, "off") == 0 ||
95 std::strcmp(opt, "no") == 0)) {
96 return false;
97 }
98 for (const char *key : {"MALLOC_TRIM_THRESHOLD_", "MALLOC_MMAP_THRESHOLD_",
99 "MALLOC_TOP_PAD_", "MALLOC_MMAP_MAX_"}) {
100 if (std::getenv(key) != nullptr) {
101 return false;
102 }
103 }
104 const char *tunables = std::getenv("GLIBC_TUNABLES");
105 if (tunables != nullptr && std::strstr(tunables, "glibc.malloc.") != nullptr) {
106 return false;
107 }

109 const int mmapAt = sizeof(void *) >= 8 ? 32 << 20 : 4 << 20;
110 mallopt(M_MMAP_THRESHOLD, mmapAt);
111 mallopt(M_TRIM_THRESHOLD, 256 << 20);

132// Rayon width for a team of `team`. Linkcell's pool steals work, so a
133// worker the scheduler parks holds up only the job it is on. Narrowed
134// with the OpenMP team beside an 8-thread spinner, the 110592-molecule
135// k-nearest search ran on two to four workers at half a cpu each and
136// took 16 to 36 ms instead of 8 to 18. The
137// pool keeps the width asked for; a slice PinSession took for another
138// copy of the program is not a narrowing and still applies.
139inline int poolWidth(int team) {
140#ifdef SEAMS_HAS_OPENMP
141 const int asked = askedWidth();
142 if (asked > team && team == omp_get_max_threads()) {
143 return asked;
144 }
145#endif
146 return team;

149// Runs `search` on the calling thread while the other OpenMP threads of
150// the team sleep on an atomic wait. After a parallel region the
151// workers spin in the default wait policy, for milliseconds on cores
152// with a slow pause instruction, and a Rayon search started then shares
153// every core with a spinning worker. Inside the region they sleep in the
154// kernel and the pool has the cores. The team stays alive, so the next
155// region does not pay the wake-up of a paused team. A mutex or condition
156// variable would throw std::system_error on failure, inside the region.
157template <class Search> void withTeamAsleep(int width, Search &&search) {
158#ifdef SEAMS_HAS_OPENMP
159 if (width < 2 || omp_in_parallel()) {
160 search();
161 return;
162 }
163 std::atomic<bool> done{false};
164 std::exception_ptr failure;
165#pragma omp parallel num_threads(width)
166 {
167 if (omp_get_thread_num() == 0) {
168 try {
169 search();
170 } catch (...) {
171 failure = std::current_exception();
172 }
173 done.store(true, std::memory_order_release);
174 done.notify_all();
175 } else {
176 done.wait(false, std::memory_order_acquire);
177 }
178 }
179 if (failure) {
180 std::rethrow_exception(failure);
181 }
182#else
183 (void)width;
184 search();
185#endif
186}

204// Team width for a request of `base` threads from the load other
205// processes put on the caller's cpus, or 0 where that load cannot be
206// read. The first look counts their runnable threads. Later looks
207// average the cpu time they took over at least 0.1 s, from /proc/stat

209// Half the cpus busy or more gives half of base; fewer busy cpus give

219// Pins the OpenMP team for the enclosing call, then puts every worker's
220// affinity back. SEAMS_PIN=0 does nothing. A host that already chose
221// places (OMP_PROC_BIND, OMP_PLACES, GOMP_CPU_AFFINITY, KMP_AFFINITY), or
222// a caller whose mask is a single cpu, is left alone. A lone team
223// narrower than the mask is not pinned. When several processes share the
224// mask, each keeps a disjoint slice and the team is narrowed to that
225// slice. Beside a few busy threads the team narrows to
226// the free cpus and is not pinned; when other threads hold half the mask
227// it is left to the scheduler and FairWidth. The outermost session calls
228// widenAffinity before a load-narrowed width can make it skip, so the
229// masks it saves and restores are the widened ones. Placement adds no cpu
230// outside the host mask. One host thread holds the placement at a time;
231// a call from another thread meanwhile runs unplaced. With team false only
232// linkcell's pool is placed: the plan and its width apply, and no OpenMP
233// region runs to pin or restore the workers, for a call that starts no
234// OpenMP team.

303// Sets the OpenMP width from teamFor for the rest of the scope, then
304// puts the caller's width back. An OpenMP-only kernel uses this instead
305// of adoptThreads, which would also start the Rayon pool.
306struct TeamWidth {
307 int saved = 1;
308 bool on = false;
309#ifdef __linux__
310 cpu_set_t oldMask;
311#endif
312 explicit TeamWidth(int n) {
313#ifdef SEAMS_HAS_OPENMP
314 if (!omp_in_parallel()) {
315 saved = std::max(1, omp_get_max_threads());
316 const int want = teamFor(n);
317 if (want != saved) {
318 omp_set_num_threads(want);
319 on = true;
320 }
321#ifdef __linux__
322 // A four-thread team on an eight-core mask still migrates across
323 // all eight. Bind it to the first cores the process is allowed,
324 // before the workers are created, so they inherit that mask.
325 if (want > 0 && want < saved &&
326 sched_getaffinity(0, sizeof(oldMask), &oldMask) == 0) {
327 // CPU_ISSET and CPU_SET take the cpu number as a size_t.
328 std::size_t cpus[256];
329 int ncpu = 0;
330 for (std::size_t cpu = 0; cpu < CPU_SETSIZE && ncpu < want && ncpu < 256; ++cpu) {
331 if (CPU_ISSET(cpu, &oldMask)) {
332 cpus[ncpu++] = cpu;
333 }
334 }
335 if (ncpu > 0 && !placementHeld()) {
336 // Once per process. Repeating the bind on a call of a few
337 // hundred microseconds was the whole gap between four threads
338 // requested directly and eight narrowed to four. OpenMP's own
339 // bind would put the workers back on the whole allowed set,
340 // so it is turned off before the team is created.
341 // Thread 0 is the caller and stays on its own mask: a pinned
342 // caller hands its one cpu to every OpenMP worker and Rayon
343 // thread it creates later, and the next eight-thread region
344 // then runs five threads on that cpu. A live PinSession already
345 // placed the team; binding again would stack that narrower
346 // team on the caller's cpu.
347 static int bound = 0;
348 static std::mutex boundMu;
349 const std::lock_guard<std::mutex> lock(boundMu);
350 if (bound != want) {
351 setenv("OMP_PROC_BIND", "false", 1);
352#pragma omp parallel num_threads(want)
353 {
354 const int tid = omp_get_thread_num();
355 if (tid != 0) {
356 cpu_set_t mask;
357 CPU_ZERO(&mask);
358 CPU_SET(cpus[tid < ncpu ? tid : 0], &mask);
359 sched_setaffinity(0, sizeof(mask), &mask);
360 }
361 }
362 bound = want;
363 }
364 }
365 }
366#endif
367 }
368#else
369 (void)n;
370#endif
371 }
372 ~TeamWidth() {
373#ifdef SEAMS_HAS_OPENMP
374 if (on) {
375 omp_set_num_threads(saved);
376 }
377#endif
378 }
379 TeamWidth(const TeamWidth &) = delete;
380 TeamWidth &operator=(const TeamWidth &) = delete;

383// Team width for one library call on a machine other processes share.
384// When their threads hold the cores, a full-width team waits at each fork
385// that follows serial work for workers the scheduler has parked, a slice
386// at a time, and a call of a few milliseconds grows tenfold. A narrower
387// team leaves cores free and does not wait. On Linux the width follows
388// the load other processes put on the caller's cpus (loadWidth). Where
389// that load cannot be read, one small region at entry, after the
390// caller's own gap, takes at most a few hundred microseconds on an idle
391// machine, where sleeping workers wake, and milliseconds on a contended
392// one. Two slow entries in a row halve the team, down to a quarter of
393// the requested width; at most once a second a narrowed team tries
394// twice its width again. Results do not depend on the width.
395// SEAMS_FAIR_WIDTH=0 keeps the requested width.
396struct FairState {
397 std::mutex lock;
398 int base = 0;
399 int granted = 0;
400 int slow = 0;
401 double tried = 0.0;
402};

446private:
447 // Constructed before the constructor body, so the probe fork is placed.
448 PinSession placed_;
449#ifdef SEAMS_HAS_OPENMP
450 static constexpr double kSlowFork = 1.0e-3;
451 // A full-width fork that waits out a time slice is a busy machine, not
452 // jitter, and halves the team at once. Beside an 8-thread spinner the
453 // bound 8-thread probe came back in microseconds one call in two or three
454 // while the call itself took 45 to 85 ms, so waiting for two slow probes
455 // in a row left a CHILL run at full width for its whole life; at four
456 // threads it took 10 ms. Going below half still takes two slow probes:
457 // two threads took 18 ms. An idle fork takes tens of microseconds.
458 static constexpr double kSliceFork = 2.0e-3;
459 static constexpr double kRetry = 1.0;
460 int asked_ = 1;
461 bool set_ = false;
462 bool marked_ = false;
463
464 static int &depth() { return fairDepth(); }
465
466 static double forkSeconds(int width) {
467 const double start = omp_get_wtime();
468#pragma omp parallel num_threads(width)
469 {
470 [[maybe_unused]] volatile int touch = omp_get_thread_num();
471 }
472 return omp_get_wtime() - start;
473 }

475 void choose() {
476 asked_ = std::max(1, omp_get_max_threads());
477 if (asked_ < 2) {
478 return;
479 }
480 FairState &state = fairState();
481 const std::lock_guard<std::mutex> hold(state.lock);
482 if (state.base != asked_) {
483 state.base = asked_;
484 state.granted = asked_;
485 state.slow = 0;
486 state.tried = 0.0;
487 }
488 const int loaded = loadWidth(state.base);
489 if (loaded > 0) {
490 state.granted = loaded;
491 } else {
492 probe(state);
493 }
494 if (state.granted != asked_) {
495 omp_set_num_threads(state.granted);
496 set_ = true;
497 if (askedWidth() == 0) {
498 askedWidth() = asked_;
499 marked_ = true;
500 }
501 }
502 }
503
504 static void probe(FairState &state) {
505 const int floor = std::max(1, state.base / 4);
506 const double now = omp_get_wtime();
507 int width = state.granted;
508 if (width < state.base && now - state.tried >= kRetry) {
509 width = std::min(state.base, 2 * width);
510 state.tried = now;
511 }
512 const double took = forkSeconds(width);
513 if (took <= kSlowFork) {
514 state.slow = 0;
515 state.granted = width;
516 } else if (width == state.granted && width > floor &&
517 (++state.slow >= 2 || (took > kSliceFork && width == state.base))) {
518 state.granted = std::max(floor, width / 2);
519 state.slow = 0;
520 state.tried = now;
521 }
522 }
523#endif
524};
