7// Solid harmonics for l = 3, 4 and 6. A bond (x, y, z) of squared length
8// r2 contributes Y_lm = S_lm(x, y, z) / r^l, and S_lm is a polynomial.
9// Even l divides by a power of r2, so the harmonic has no square root.
10// l = 3 keeps one factor 1/(r2 * sqrt(r2)).
11//
12// phi = atan2(x, y), so (y + i x)^m carries sin(theta)^m exp(i m phi).
13// The caller passes inv = 1/r^l. A zero bond passes inv = 0 and a zero
14// vector; the polynomials then contribute nothing.
15

19template <class T>
20inline void eval(int orderL, T x, T y, T z, T r2, T inv, const double *norm,
21 T *re, T *im) {
22 T q0 = T(0);
23 T q1 = T(0);
24 T q2 = T(0);
25 T q3 = T(0);
26 T q4 = T(0);
27 T q5 = T(0);
28 T q6 = T(0);
29 const T z2 = z * z;
30 if (orderL == 3) {
31 q0 = z * (T(5) * z2 - T(3) * r2);
32 q1 = T(5) * z2 - r2;
33 q2 = z;
34 q3 = T(1);
35 } else if (orderL == 4) {
36 const T z3 = z2 * z;
37 const T z4 = z2 * z2;
38 const T r4 = r2 * r2;
39 q0 = T(35) * z4 - T(30) * z2 * r2 + T(3) * r4;
40 q1 = T(7) * z3 - T(3) * z * r2;
41 q2 = T(7) * z2 - r2;
42 q3 = z;
43 q4 = T(1);
44 } else {
45 const T z3 = z2 * z;
46 const T z4 = z2 * z2;
47 const T z5 = z4 * z;
48 const T z6 = z4 * z2;
49 const T r4 = r2 * r2;
50 const T r6 = r4 * r2;
51 q0 = T(231) * z6 - T(315) * z4 * r2 + T(105) * z2 * r4 - T(5) * r6;
52 q1 = T(33) * z5 - T(30) * z3 * r2 + T(5) * z * r4;
53 q2 = T(33) * z4 - T(18) * z2 * r2 + r4;
54 q3 = T(11) * z3 - T(3) * z * r2;
55 q4 = T(11) * z2 - r2;
56 q5 = z;
57 q6 = T(1);
58 }
59
60 T q[7];
61 q[0] = q0;
62 q[1] = q1;
63 q[2] = q2;
64 q[3] = q3;
65 if (orderL >= 4) {
66 q[4] = q4;
67 }
68 if (orderL == 6) {
69 q[5] = q5;
70 q[6] = q6;
71 }
72
73 T pr = T(1);
74 T pi = T(0);
75 const T a0 = T(norm[0]) * q[0] * inv;
76 re[orderL] = a0;
77 im[orderL] = T(0);
78 for (int m = 1; m <= orderL; ++m) {
79 const T npr = pr * y - pi * x;
80 const T npi = pr * x + pi * y;
81 pr = npr;
82 pi = npi;
83 const T amp = T(norm[m]) * q[m] * inv;
84 const int ineg = orderL - m;
85 const int ipos = orderL + m;
86 re[ineg] = amp * pr;
87 im[ineg] = -amp * pi;
88 const T sign = (m & 1) ? T(-1) : T(1);
89 re[ipos] = sign * amp * pr;
90 im[ipos] = sign * amp * pi;
91 }
92}
