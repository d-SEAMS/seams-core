557inline void check(int status, const void *handle) {
558 if (status == SEAMS_OK) {
559 return;
560 }
561 const char *text = seams_handle_error(handle);
562 throw Error(status, text == nullptr ? std::string() : std::string(text));

565template <class T, void (*Destroy)(T *)>
566class Handle {
567 public:
568 Handle() = default;
569 explicit Handle(T *ptr) noexcept : ptr_(ptr) {}
570 Handle(const Handle &) = delete;
571 Handle &operator=(const Handle &) = delete;
572 Handle(Handle &&other) noexcept : ptr_(std::exchange(other.ptr_, nullptr)) {}
573 Handle &operator=(Handle &&other) noexcept {
574 if (this != &other) {
575 reset();
576 ptr_ = std::exchange(other.ptr_, nullptr);
577 }
578 return *this;
579 }
580 ~Handle() { reset(); }
581
582 [[nodiscard]] explicit operator bool() const noexcept { return ptr_ != nullptr; }
583
584 protected:
585 [[nodiscard]] T *get() const noexcept { return ptr_; }
586
587 void reset() noexcept {
588 if (ptr_ != nullptr) {
589 Destroy(ptr_);
590 ptr_ = nullptr;
591 }
592 }
593 T *ptr_ = nullptr;
594};

659 [[nodiscard]] std::vector<int> degrees() const {
660 std::size_t n_offsets = 0;
661 std::size_t count = 0;
662 detail::check(seams_neighbours_copy(get(), nullptr, 0, &n_offsets, nullptr, 0, &count), get());
663 std::vector<int> offsets(n_offsets);
664 std::vector<int> cols(count);
665 detail::check(seams_neighbours_copy(get(), offsets.data(), offsets.size(), &n_offsets,
666 cols.data(), cols.size(), &count),
667 get());
668 std::vector<int> degree;
669 if (n_offsets > 0) {
670 degree.reserve(n_offsets - 1);
671 for (std::size_t i = 0; i + 1 < n_offsets; ++i) {
672 degree.push_back(offsets[i + 1] - offsets[i]);
673 }
674 }
675 return degree;
676 }

682 [[nodiscard]] std::vector<std::pair<int, int>> pairs() const {
683 std::size_t n_offsets = 0;
684 std::size_t count = 0;
685 detail::check(seams_neighbours_copy(get(), nullptr, 0, &n_offsets, nullptr, 0, &count), get());
686 std::vector<int> offsets(n_offsets);
687 std::vector<int> cols(count);
688 detail::check(seams_neighbours_copy(get(), offsets.data(), offsets.size(), &n_offsets,
689 cols.data(), cols.size(), &count),
690 get());
691 std::vector<std::pair<int, int>> out;
692 out.reserve(count);
693 for (std::size_t atom = 0; atom + 1 < n_offsets; ++atom) {
694 for (int cursor = offsets[atom]; cursor < offsets[atom + 1]; ++cursor) {
695 const int neighbour = cols[static_cast<std::size_t>(cursor)];
696 if (neighbour != static_cast<int>(atom)) {
697 out.emplace_back(static_cast<int>(atom), neighbour);
698 }
699 }
700 }
701 return out;
702 }
