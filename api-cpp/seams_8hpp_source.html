802inline void check(int status, const void *handle) {
803 if (status == SEAMS_OK) {
804 return;
805 }
806 const char *text = seams_handle_error(handle);
807 throw Error(status, text == nullptr ? std::string() : std::string(text));

810template <class T, void (*Destroy)(T *)>
811class Handle {
812 public:
813 Handle() = default;
814 explicit Handle(T *ptr) noexcept : ptr_(ptr) {}
815 Handle(const Handle &) = delete;
816 Handle &operator=(const Handle &) = delete;
817 Handle(Handle &&other) noexcept : ptr_(std::exchange(other.ptr_, nullptr)) {}
818 Handle &operator=(Handle &&other) noexcept {
819 if (this != &other) {
820 reset();
821 ptr_ = std::exchange(other.ptr_, nullptr);
822 }
823 return *this;
824 }
825 ~Handle() { reset(); }
826
827 [[nodiscard]] explicit operator bool() const noexcept { return ptr_ != nullptr; }
828
829 protected:
830 [[nodiscard]] T *get() const noexcept { return ptr_; }
831
832 void reset() noexcept {
833 if (ptr_ != nullptr) {
834 Destroy(ptr_);
835 ptr_ = nullptr;
836 }
837 }
838 T *ptr_ = nullptr;
839};

841// The C trajectory calls take no xyz length and read nframes * 3 * natoms.
842inline std::size_t trajectory_coordinates(std::span<const double> xyz, std::size_t nframes,
843 std::size_t natoms) {
844 constexpr std::size_t most = std::numeric_limits<std::size_t>::max();
845 if (natoms > most / 3 || (natoms != 0 && nframes > most / (3 * natoms))) {
846 throw Error(SEAMS_ERR_RANGE, "nframes * 3 * natoms overflows size_t");
847 }
848 const std::size_t want = nframes * 3 * natoms;
849 if (xyz.size() != want) {
850 throw Error(SEAMS_ERR_RANGE, "xyz length is not nframes * 3 * natoms");
851 }
852 return want;
853}

941 [[nodiscard]] std::vector<std::pair<int, int>> pairs() const {
942 std::size_t n_offsets = 0;
943 std::size_t count = 0;
944 detail::check(seams_neighbours_copy(get(), nullptr, 0, &n_offsets, nullptr, 0, &count), get());
945 std::vector<int> offsets(n_offsets);
946 std::vector<int> cols(count);
947 detail::check(seams_neighbours_copy(get(), offsets.data(), offsets.size(), &n_offsets,
948 cols.data(), cols.size(), &count),
949 get());
950 std::vector<std::pair<int, int>> out;
951 out.reserve(count);
952 for (std::size_t atom = 0; atom + 1 < n_offsets; ++atom) {
953 for (int cursor = offsets[atom]; cursor < offsets[atom + 1]; ++cursor) {
954 const int neighbour = cols[static_cast<std::size_t>(cursor)];
955 if (neighbour != static_cast<int>(atom)) {
956 out.emplace_back(static_cast<int>(atom), neighbour);
957 }
958 }
959 }
960 return out;
