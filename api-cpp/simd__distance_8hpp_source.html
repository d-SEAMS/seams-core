22// Compute squared periodic distances for a batch of atom pairs.
23// Orthorhombic lengths only: one independent wrap per axis.
24// A LAMMPS dump with tilt (PointCloud.box.size() >= 6) stores bound
25// spans in box[0..2], not lx, ly, lz. Call gen::periodicDistSq or
26// gen::batchPeriodicDistSq for those clouds.
27// dx, dy, dz: coordinate differences (length n)
28// bx, by, bz: periodic box dimensions
29// out: output squared distances (length n)
30inline HWY_ATTR void BatchPeriodicDistSq(const double* HWY_RESTRICT dx,
31 const double* HWY_RESTRICT dy,
32 const double* HWY_RESTRICT dz,
33 double bx, double by, double bz,
34 double* HWY_RESTRICT out, size_t n) {
35 const hn::ScalableTag<double> d;
36 const size_t N = hn::Lanes(d);
37
38 const auto vbx = hn::Set(d, bx);
39 const auto vby = hn::Set(d, by);
40 const auto vbz = hn::Set(d, bz);
41
42 // The box is fixed across the batch, so the three divisions of the minimum
43 // image convention collapse into one reciprocal each, hoisted out of the
44 // loop. Vector division has several times the latency of multiplication.
45 const double rbx = 1.0 / bx;
46 const double rby = 1.0 / by;
47 const double rbz = 1.0 / bz;
48 const auto vrbx = hn::Set(d, rbx);
49 const auto vrby = hn::Set(d, rby);
50 const auto vrbz = hn::Set(d, rbz);
51
52 size_t i = 0;
53 for (; i + N <= n; i += N) {
54 // Callers pass std::vector<double> scratch (16-byte typical). Load/Store
55 // require native vector alignment and fault under AVX2/AVX-512.
56 auto vdx = hn::LoadU(d, dx + i);
57 auto vdy = hn::LoadU(d, dy + i);
58 auto vdz = hn::LoadU(d, dz + i);
59
60 // Absolute values
61 vdx = hn::Abs(vdx);
62 vdy = hn::Abs(vdy);
63 vdz = hn::Abs(vdz);
64
65 // Periodic wrap: dr -= box * round(dr * (1 / box))
66 vdx = hn::NegMulAdd(vbx, hn::Round(hn::Mul(vdx, vrbx)), vdx);
67 vdy = hn::NegMulAdd(vby, hn::Round(hn::Mul(vdy, vrby)), vdy);
68 vdz = hn::NegMulAdd(vbz, hn::Round(hn::Mul(vdz, vrbz)), vdz);
69
70 // r2 = dx*dx + dy*dy + dz*dz
71 auto r2 = hn::MulAdd(vdx, vdx, hn::MulAdd(vdy, vdy, hn::Mul(vdz, vdz)));
72 hn::StoreU(r2, d, out + i);
73 }
74
75 // Scalar remainder
76 for (; i < n; i++) {
77 double ddx = std::fabs(dx[i]);
78 double ddy = std::fabs(dy[i]);
79 double ddz = std::fabs(dz[i]);
80 ddx -= bx * std::round(ddx * rbx);
81 ddy -= by * std::round(ddy * rby);
82 ddz -= bz * std::round(ddz * rbz);
83 out[i] = ddx * ddx + ddy * ddy + ddz * ddz;
84 }
85}
