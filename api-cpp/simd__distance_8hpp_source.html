26// Compute squared periodic distances for a batch of atom pairs.
27// Orthorhombic lengths only: one independent wrap per axis.
28// A LAMMPS dump with tilt (PointCloud.box.size() >= 6) stores bound
29// spans in box[0..2], not lx, ly, lz. Call gen::periodicDistSq or
30// gen::batchPeriodicDistSq for those clouds.
31// dx, dy, dz: coordinate differences (length n)
32// bx, by, bz: periodic box dimensions
33// out: output squared distances (length n)
34inline HWY_ATTR void BatchPeriodicDistSq(const double* HWY_RESTRICT dx,
35 const double* HWY_RESTRICT dy,
36 const double* HWY_RESTRICT dz,
37 double bx, double by, double bz,
38 double* HWY_RESTRICT out, size_t n) {
39#ifdef SEAMS_HAS_MINIMAGE
40 if (mi_dist2_ortho_diffs(dx, dy, dz, bx, by, bz, out, n) == 0) {
41 return;
42 }
43#endif
44 const hn::ScalableTag<double> d;
45 const size_t N = hn::Lanes(d);
46
47 const auto vbx = hn::Set(d, bx);
48 const auto vby = hn::Set(d, by);
49 const auto vbz = hn::Set(d, bz);
50
51 // The box is fixed across the batch, so the three divisions of the minimum
52 // image convention collapse into one reciprocal each, hoisted out of the
53 // loop. Vector division has several times the latency of multiplication.
54 const double rbx = 1.0 / bx;
55 const double rby = 1.0 / by;
56 const double rbz = 1.0 / bz;
57 const auto vrbx = hn::Set(d, rbx);
58 const auto vrby = hn::Set(d, rby);
59 const auto vrbz = hn::Set(d, rbz);
60
61 size_t i = 0;
62 for (; i + N <= n; i += N) {
63 // Callers pass std::vector<double> scratch (16-byte typical). Load/Store
64 // require native vector alignment and fault under AVX2/AVX-512.
65 auto vdx = hn::LoadU(d, dx + i);
66 auto vdy = hn::LoadU(d, dy + i);
67 auto vdz = hn::LoadU(d, dz + i);
68
69 // Absolute values
70 vdx = hn::Abs(vdx);
71 vdy = hn::Abs(vdy);
72 vdz = hn::Abs(vdz);
73
74 // Periodic wrap: dr -= box * round(dr * (1 / box))
75 vdx = hn::NegMulAdd(vbx, hn::Round(hn::Mul(vdx, vrbx)), vdx);
76 vdy = hn::NegMulAdd(vby, hn::Round(hn::Mul(vdy, vrby)), vdy);
77 vdz = hn::NegMulAdd(vbz, hn::Round(hn::Mul(vdz, vrbz)), vdz);
78
79 // r2 = dx*dx + dy*dy + dz*dz
80 auto r2 = hn::MulAdd(vdx, vdx, hn::MulAdd(vdy, vdy, hn::Mul(vdz, vdz)));
81 hn::StoreU(r2, d, out + i);
82 }
83
84 // Scalar remainder
85 for (; i < n; i++) {
86 double ddx = std::fabs(dx[i]);
87 double ddy = std::fabs(dy[i]);
88 double ddz = std::fabs(dz[i]);
89 ddx -= bx * std::round(ddx * rbx);
90 ddy -= by * std::round(ddy * rby);
91 ddz -= bz * std::round(ddz * rbz);
92 out[i] = ddx * ddx + ddy * ddy + ddz * ddz;
93 }
94}

96} // namespace seams
