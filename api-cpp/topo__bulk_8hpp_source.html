257class NeighBond {
258public:
259 void build(const std::vector<std::vector<int>> &nList);
260
261 // Row i, sorted and unique. Empty when i is outside the cloud.
262 void row(int i, const int *&begin, const int *&end) const {
263 if (static_cast<unsigned>(i) >= static_cast<unsigned>(n_) || off_.empty()) {
264 begin = nullptr;
265 end = nullptr;
266 return;
267 }
268 begin = cols_.data() + off_[static_cast<std::size_t>(i)];
269 end = cols_.data() + off_[static_cast<std::size_t>(i + 1)];
270 }
271
272 [[nodiscard]] bool operator()(int i, int j) const {
273 const int *begin = nullptr;
274 const int *end = nullptr;
275 row(i, begin, end);
276 if (begin == nullptr) {
277 return false;
278 }
279 return std::binary_search(begin, end, j);
280 }
281
282private:
283 int n_ = 0;
284 // Row i is cols_[off_[i] .. off_[i + 1]], sorted and unique.
285 std::vector<int> off_;
286 std::vector<int> cols_;
287};
