158struct EquatorChoice {
159 int ring[6]{};
160 int n = 0;
161 std::vector<int> past;
162
163 [[nodiscard]] int idAt(int k) const {
164 return k < 6 ? ring[k] : past[static_cast<std::size_t>(k - 6)];
165 }
166
167 void put(int k, int value) {
168 if (k < 6) {
169 ring[k] = value;
170 return;
171 }
172 const std::size_t at = static_cast<std::size_t>(k - 6);
173 if (past.size() <= at) {
174 past.resize(at + 1);
175 }
176 past[at] = value;
177 }
178};
