268class NeighBond {
269public:
270 void build(const std::vector<std::vector<int>> &nList);
271
272 // Row i, sorted and unique. Empty when i is outside the cloud.
273 void row(int i, const int *&begin, const int *&end) const {
274 if (static_cast<unsigned>(i) >= static_cast<unsigned>(n_) || off_.empty()) {
275 begin = nullptr;
276 end = nullptr;
277 return;
278 }
279 begin = cols_.data() + off_[static_cast<std::size_t>(i)];
280 end = cols_.data() + off_[static_cast<std::size_t>(i + 1)];
281 }
282
283 [[nodiscard]] bool operator()(int i, int j) const {
284 const int *begin = nullptr;
285 const int *end = nullptr;
286 row(i, begin, end);
287 if (begin == nullptr) {
288 return false;
289 }
290 return std::binary_search(begin, end, j);
291 }
292
293private:
294 int n_ = 0;
295 // Row i is cols_[off_[i] .. off_[i + 1]], sorted and unique.
296 std::vector<int> off_;
297 std::vector<int> cols_;
298};
